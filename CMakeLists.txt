cmake_minimum_required(VERSION 3.24)
project(HPLJ1005x64 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Keep the printer DLLs usable on computers without the Visual C++ redistributable.
# All code linked into a module uses the same runtime; exported interfaces use
# Windows/COM ownership rules rather than passing CRT allocations across modules.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This project targets Windows x64; configure with -A x64")
endif()
add_library(jbig STATIC vendor/jbig/jbig.c vendor/jbig/jbig_ar.c)
target_include_directories(jbig PUBLIC vendor/jbig)
add_library(lj1005_core STATIC src/protocol.cpp src/firmware.cpp src/pjl.cpp)
target_include_directories(lj1005_core PUBLIC include)
target_link_libraries(lj1005_core PUBLIC jbig)
add_executable(lj1005 src/main.cpp)
target_link_libraries(lj1005 PRIVATE lj1005_core)
if(WIN32)
  add_library(lj1005monitor SHARED driver/monitor.cpp driver/trace.cpp driver/completion_windows.cpp src/usb_windows.cpp driver/monitor.def)
  target_link_libraries(lj1005monitor PRIVATE lj1005_core winspool setupapi cfgmgr32 advapi32)
  target_compile_definitions(lj1005monitor PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(lj1005monitor PRIVATE /W4 /WX /permissive- /EHsc)
  endif()
  add_executable(queue_test src/queue_test.cpp)
  target_compile_definitions(queue_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(queue_test PRIVATE gdi32 winspool)
  if(MSVC)
    target_compile_options(queue_test PRIVATE /W4 /WX /EHsc)
  endif()
  add_library(lj1005render SHARED driver/render.cpp driver/render.def)
  target_link_libraries(lj1005render PRIVATE lj1005_core ole32 uuid)
  target_compile_definitions(lj1005render PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(lj1005render PRIVATE /W4 /WX /permissive- /EHsc)
  endif()
  add_custom_target(driver_package ALL
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/driver/lj1005.gpd" "${CMAKE_CURRENT_SOURCE_DIR}/driver/lj1005.inf" "${CMAKE_CURRENT_SOURCE_DIR}/driver/lj1005.ini"
      "$<TARGET_FILE_DIR:lj1005render>"
    DEPENDS lj1005render lj1005monitor)
  target_sources(lj1005 PRIVATE src/usb_windows.cpp)
  target_link_libraries(lj1005 PRIVATE setupapi cfgmgr32)
  target_compile_definitions(lj1005 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  target_compile_options(lj1005_core PRIVATE /W4 /WX /permissive- /EHsc)
  target_compile_options(lj1005 PRIVATE /W4 /WX /permissive- /EHsc)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(pjl_tests tests/pjl_tests.cpp)
  target_link_libraries(pjl_tests PRIVATE lj1005_core)
  if(MSVC)
    target_compile_options(pjl_tests PRIVATE /W4 /WX /EHsc)
  endif()
  add_test(NAME pjl_status COMMAND pjl_tests)
  add_executable(firmware_tests tests/firmware_tests.cpp)
  target_link_libraries(firmware_tests PRIVATE lj1005_core)
  if(MSVC)
    target_compile_options(firmware_tests PRIVATE /W4 /WX /EHsc)
  endif()
  add_test(NAME firmware COMMAND firmware_tests)
  if(WIN32)
    add_library(monitor_fixture SHARED driver/monitor.cpp driver/trace.cpp tests/completion_fixture.cpp driver/monitor.def)
    target_link_libraries(monitor_fixture PRIVATE lj1005_core advapi32)
    target_compile_definitions(monitor_fixture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(monitor_fixture PRIVATE /W4 /WX /EHsc)
    endif()
    add_executable(monitor_tests tests/monitor_tests.cpp)
    target_compile_definitions(monitor_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
      target_compile_options(monitor_tests PRIVATE /W4 /WX /EHsc)
    endif()
    add_test(NAME monitor_callbacks COMMAND monitor_tests "$<TARGET_FILE:monitor_fixture>")
    add_test(NAME monitor_production_load COMMAND monitor_tests "$<TARGET_FILE:lj1005monitor>" --load-only)
    add_executable(render_smoke tests/render_smoke.cpp)
    target_compile_definitions(render_smoke PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(render_smoke PRIVATE /EHsc)
    target_link_libraries(render_smoke PRIVATE lj1005_core ole32 uuid)
    add_test(NAME render_com COMMAND render_smoke "$<TARGET_FILE:lj1005render>")
  endif()
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE lj1005_core)
  if(MSVC)
    target_compile_options(protocol_tests PRIVATE /EHsc)
  endif()
  add_test(NAME protocol COMMAND protocol_tests)
  if(MSVC AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/foo2zjs/foo2zjs.c")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/reference")
    file(READ third_party/foo2zjs/zjs.h ref_header)
    string(REPLACE "(( SWAP_32 )dword).byte" "((char*)&dword)" ref_header "${ref_header}")
    string(REPLACE "(( SWAP_16 )word).byte" "((char*)&word)" ref_header "${ref_header}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reference/zjs.h" "${ref_header}")
    foreach(ref foo2zjs arm2hpdl)
      file(READ "third_party/foo2zjs/${ref}.c" ref_source)
      string(REPLACE "\"r\"" "\"rb\"" ref_source "${ref_source}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reference/${ref}.c" "${ref_source}")
      add_library(${ref}_source OBJECT "${CMAKE_CURRENT_BINARY_DIR}/reference/${ref}.c")
      target_compile_definitions(${ref}_source PRIVATE main=reference_main _CRT_SECURE_NO_WARNINGS)
      target_include_directories(${ref}_source PRIVATE tests/compat third_party/foo2zjs)
      add_executable(reference_${ref} tests/reference_main.c $<TARGET_OBJECTS:${ref}_source>)
      target_link_libraries(reference_${ref} PRIVATE jbig)
    endforeach()
  endif()
endif()
