cmake_minimum_required(VERSION 3.24)
project(HPLJ1005x64 VERSION 0.4.0 LANGUAGES C CXX)
set(OPENZJS_VERSION "${PROJECT_VERSION}.0")
# Windows stores each version component as a 16-bit integer.
foreach(component IN ITEMS PROJECT_VERSION_MAJOR PROJECT_VERSION_MINOR PROJECT_VERSION_PATCH)
  if(${component} GREATER 65535)
    message(FATAL_ERROR "Version components must fit in 16 bits")
  endif()
endforeach()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/driver/lj1005.inf" inf_version_line REGEX "^DriverVer=")
# The date is maintained with the INF; only the version is tied to CMake.
string(REGEX REPLACE "^DriverVer=[^,]+," "" inf_version "${inf_version_line}")
if(NOT inf_version STREQUAL OPENZJS_VERSION)
  message(FATAL_ERROR "CMake and INF versions differ")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Keep the printer DLLs usable on computers without the Visual C++ redistributable.
# All code linked into a module uses the same runtime; exported interfaces use
# Windows/COM ownership rules rather than passing CRT allocations across modules.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This project targets Windows x64; configure with -A x64")
endif()

target_link_libraries(lj1005 PRIVATE lj1005_core)

    set(OPENZJS_FILE_DESCRIPTION "${description}")
    set(OPENZJS_ORIGINAL_FILENAME "${original_filename}")
    set(OPENZJS_INTERNAL_NAME "${target}")
    set(OPENZJS_FILE_TYPE "${file_type}")
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/driver/version.rc.in"
                   "${CMAKE_CURRENT_BINARY_DIR}/generated/${target}.rc" @ONLY)
    target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/${target}.rc")
  endfunction()
